        // ---------------- NAVIGATION LOGIC ----------------
        function navigateTo(pageId) {
            // Hide all pages
            document.querySelectorAll('.page-section').forEach(section => {
                section.classList.remove('active');
            });

            // Show selected page
            const target = document.getElementById(`page-${pageId}`);
            if (target) {
                target.classList.add('active');
            }

            // Update Navbar Active State Indicators
            document.querySelectorAll('.nav-btn').forEach(btn => {
                const indicator = btn.querySelector('.nav-indicator');
                if (btn.id === `nav-${pageId}`) {
                    btn.classList.add('text-white');
                    btn.classList.remove('text-slate-200');
                    if (indicator) indicator.classList.remove('opacity-0');
                } else {
                    btn.classList.remove('text-white');
                    btn.classList.add('text-slate-200');
                    if (indicator) indicator.classList.add('opacity-0');
                }
            });

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function filterExamples(category) {

            tbody.innerHTML = '';

            // Update UI buttons
            document.querySelectorAll('.filter-btn').forEach(btn => {

                const chev = activeBtn.querySelector('.fa-chevron-right');
                if (chev) chev.classList.remove('opacity-0');

            if (category === 'all') {
                titleEl.innerText = 'Semua Contoh Kalimat';
                Object.values(examplesData).forEach(arr => rows.push(...arr));
            } else {

                    statements: 'Contoh Statements (Pernyataan)',
                    commands: 'Contoh Commands & Requests',

                    nobackshift: 'Contoh No Backshift'

                rows = examplesData[category] || [];

                const tr = document.createElement('tr');
                tr.className = idx % 2 === 0 ? 'bg-white' : 'bg-slate-50/50';
                tr.innerHTML = `
                    <td class="p-4 pl-6 text-slate-800">${item.direct}</td>
                    <td class="p-4 pr-6 text-navy-800 font-semibold">${item.indirect}</td>
                `;
                tbody.appendChild(tr);
            });
        }

            document.querySelectorAll('.quiz-container').forEach(el => el.classList.add('hidden'));

            if (mode === 'mcq') {
                document.getElementById('quiz-mode-mcq').classList.remove('hidden');
                renderMCQ();
            } else if (mode === 'transform') {
                document.getElementById('quiz-mode-transform').classList.remove('hidden');
                renderTrans();
            } else if (mode === 'review') {
                document.getElementById('quiz-mode-review').classList.remove('hidden');
                renderReview();
            }
        }

        // --- MCQ Functions ---
        function renderMCQ() {
            const q = mcqQuestions[currentMCQIndex];
            document.getElementById('mcq-question-number').innerText = `${currentMCQIndex + 1}. Pilih jawaban yang paling tepat!`;
            document.getElementById('mcq-question-text').innerText = q.q;

            const container = document.getElementById('mcq-options-container');
            container.innerHTML = '';

            const labels = ['A', 'B', 'C', 'D'];
            q.options.forEach((opt, idx) => {
                const isSelected = userMCQAnswers[currentMCQIndex] === idx;
                const label = document.createElement('label');
                label.className = `flex items-center gap-3 p-4 rounded-xl border transition-all cursor-pointer font-medium ${
                    isSelected ? 'border-navy-800 bg-sky-soft text-navy-800 ring-2 ring-navy-800/20' : 'border-slate-200 hover:bg-slate-50 text-slate-700'
                }`;
                label.innerHTML = `
                    <input type="radio" name="mcq-option" value="${idx}" ${isSelected ? 'checked' : ''} onchange="selectMCQOption(${idx})" class="w-4 h-4 text-navy-800 focus:ring-navy-800">
                    <span class="font-bold">${labels[idx]}.</span>
                    <span class="text-sm sm:text-base">${opt}</span>
                `;
                container.appendChild(label);
            });

            // Update Progress Counter & Bar
            const total = mcqQuestions.length;
            const currentNum = currentMCQIndex + 1;
            document.getElementById('quiz-progress-counter').innerText = `${currentNum}/${total}`;
            document.getElementById('quiz-progress-bar').style.width = `${(currentNum / total) * 100}%`;

            // Update Nav Buttons
            document.getElementById('btn-mcq-prev').disabled = (currentMCQIndex === 0);
            document.getElementById('btn-mcq-prev').classList.toggle('opacity-40', currentMCQIndex === 0);
        }

        function selectMCQOption(idx) {
            userMCQAnswers[currentMCQIndex] = idx;
            renderMCQ();
        }

        function nextMCQ() {
            if (currentMCQIndex < mcqQuestions.length - 1) {
                currentMCQIndex++;
                renderMCQ();
            } else {
                // Show completion summary modal or jump to review
                switchQuizMode('review');
            }
        }

        function prevMCQ() {
            if (currentMCQIndex > 0) {
                currentMCQIndex--;
                renderMCQ();
            }
        }

        // --- Transformation Functions ---
        function renderTrans() {
            const item = transQuestions[currentTransIndex];
            document.getElementById('trans-number').innerText = `Latihan Ubah Kalimat (${currentTransIndex + 1}/${transQuestions.length}):`;
            document.getElementById('trans-direct-text').innerText = item.direct;
            document.getElementById('trans-user-input').value = '';
            document.getElementById('trans-feedback').classList.add('hidden');

            document.getElementById('btn-trans-prev').disabled = (currentTransIndex === 0);
            document.getElementById('btn-trans-prev').classList.toggle('opacity-40', currentTransIndex === 0);
        }

        function checkTransformation() {
            const userText = document.getElementById('trans-user-input').value.trim().toLowerCase().replace(/[.,\/#!$%\^&\*;:{}=\-_`~()]/g,"");
            const targetText = transQuestions[currentTransIndex].target.toLowerCase().replace(/[.,\/#!$%\^&\*;:{}=\-_`~()]/g,"");

            const feedback = document.getElementById('trans-feedback');
            feedback.classList.remove('hidden');

            if (userText === targetText) {
                feedback.className = "p-4 rounded-xl bg-emerald-50 border border-emerald-200 text-emerald-800 font-semibold";
                feedback.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-600 mr-2"></i> Sangat Tepat! Jawaban kamu benar.`;
            } else {
                feedback.className = "p-4 rounded-xl bg-rose-50 border border-rose-200 text-rose-800 font-medium";
                feedback.innerHTML = `
                    <p class="font-bold text-rose-900"><i class="fa-solid fa-circle-xmark mr-2"></i> Kurang Tepat.</p>
                    <p class="mt-1 text-xs">Jawaban acuan: <span class="font-semibold">${transQuestions[currentTransIndex].target}</span></p>
                `;
            }
        }

        function nextTrans() {
            if (currentTransIndex < transQuestions.length - 1) {
                currentTransIndex++;
                renderTrans();
            }
        }

        function prevTrans() {
            if (currentTransIndex > 0) {
                currentTransIndex--;
                renderTrans();
            }
        }

        // --- Review Pembahasan Functions ---
        function renderReview() {
            const container = document.getElementById('review-list-container');
            container.innerHTML = '';

            let score = 0;

            mcqQuestions.forEach((q, idx) => {
                const userChoice = userMCQAnswers[idx];
                const isCorrect = userChoice === q.correct;
                if (isCorrect) score++;

                const card = document.createElement('div');
                card.className = "p-4 rounded-xl border border-slate-200 bg-slate-50 space-y-2 text-sm";
                
                const labels = ['A', 'B', 'C', 'D'];
                const userText = userChoice !== null ? `${labels[userChoice]}. ${q.options[userChoice]}` : 'Belum dijawab';
                const correctText = `${labels[q.correct]}. ${q.options[q.correct]}`;

                card.innerHTML = `
                    <div class="flex items-start justify-between gap-2">
                        <span class="font-bold text-navy-800">${idx + 1}. ${q.q}</span>
                        ${userChoice !== null ? 
                            (isCorrect ? '<span class="px-2.5 py-1 bg-emerald-100 text-emerald-700 font-extrabold text-xs rounded-md">Benar</span>' : '<span class="px-2.5 py-1 bg-rose-100 text-rose-700 font-extrabold text-xs rounded-md">Salah</span>')
                            : '<span class="px-2.5 py-1 bg-slate-200 text-slate-600 font-bold text-xs rounded-md">Kosong</span>'
                        }
                    </div>
                    <div class="text-xs space-y-1">
                        <p class="text-slate-600">Jawaban Anda: <span class="font-bold ${isCorrect ? 'text-emerald-700' : 'text-rose-600'}">${userText}</span></p>
                        <p class="text-emerald-700 font-bold">Jawaban Benar: ${correctText}</p>
                    </div>
                    <div class="p-2.5 bg-white rounded-lg border border-slate-200 text-xs text-slate-600 mt-2">
                        <strong>Pembahasan:</strong> ${q.explanation}
                    </div>
                `;
                container.appendChild(card);
            });
        }


        // ---------------- LIVE SEARCH LOGIC ----------------
        function openSearchModal() {
            document.getElementById('search-modal').classList.remove('hidden');
            document.getElementById('search-input').focus();
        }

        function closeSearchModal() {
            document.getElementById('search-modal').classList.add('hidden');
        }

        function handleLiveSearch() {
            const query = document.getElementById('search-input').value.trim().toLowerCase();
            const container = document.getElementById('search-results');

            if (query === '') {
                container.innerHTML = `<p class="text-sm text-slate-400 text-center py-6">Ketik kata kunci untuk mulai mencari...</p>`;
                return;
            }

            let matches = [];

            // Search Materi
            Object.keys(materiData).forEach(key => {
                const topic = materiData[key];
                if (topic.title.toLowerCase().includes(query) || topic.subtitle.toLowerCase().includes(query)) {
                    matches.push({ type: 'materi', id: key, title: topic.title, desc: topic.subtitle });
                }
            });

            // Search Examples
            Object.keys(examplesData).forEach(cat => {
                examplesData[cat].forEach(ex => {
                    if (ex.direct.toLowerCase().includes(query) || ex.indirect.toLowerCase().includes(query)) {
                        matches.push({ type: 'contoh', cat: cat, title: ex.direct, desc: `Indirect: ${ex.indirect}` });
                    }
                });
            });

            if (matches.length === 0) {
                container.innerHTML = `<p class="text-sm text-slate-400 text-center py-6">Tidak ada hasil yang ditemukan untuk "${query}".</p>`;
                return;
            }

            container.innerHTML = '';
            matches.forEach(m => {
                const item = document.createElement('div');
                item.className = "p-3 rounded-xl hover:bg-sky-50 transition cursor-pointer flex items-center justify-between";
                if (m.type === 'materi') {
                    item.onclick = () => { closeSearchModal(); navigateTo('materi'); openMateriTopic(m.id); };
                    item.innerHTML = `
                        <div>
                            <span class="text-xs font-bold text-blue-600 uppercase">Materi ${m.id}</span>
                            <h4 class="font-bold text-slate-900 text-sm">${m.title}</h4>
                            <p class="text-xs text-slate-500">${m.desc}</p>
                        </div>
                        <i class="fa-solid fa-chevron-right text-slate-300 text-xs"></i>
                    `;
                } else {
                    item.onclick = () => { closeSearchModal(); navigateTo('contoh'); filterExamples(m.cat); };
                    item.innerHTML = `
                        <div>
                            <span class="text-xs font-bold text-emerald-600 uppercase">Contoh Kalimat</span>
                            <h4 class="font-bold text-slate-900 text-sm">${m.title}</h4>
                            <p class="text-xs text-slate-500">${m.desc}</p>
                        </div>
                        <i class="fa-solid fa-chevron-right text-slate-300 text-xs"></i>
                    `;
                }
                container.appendChild(item);
            });
        }


        // ---------------- INITIALIZATION ----------------
        window.addEventListener('DOMContentLoaded', () => {
            // Set initial state
            filterExamples('all');
            renderMCQ();

            // Keyboard shortcut Escape for Modal
            window.addEventListener('keydown', (e) => {
                if (e.key === 'Escape') {
                    closeMateriModal();
                    closeSearchModal();
                }
            });
        });
